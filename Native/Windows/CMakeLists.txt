cmake_minimum_required(VERSION 3.20)
project(PulseVideoNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "PulseVideoNative requires Windows x64")
endif()
set(UNITY_PLUGIN_API "" CACHE PATH "Unity Editor/Data/PluginAPI directory")
if(NOT EXISTS "${UNITY_PLUGIN_API}/IUnityGraphicsD3D11.h")
    message(FATAL_ERROR "Set UNITY_PLUGIN_API to the installed Unity Editor/Data/PluginAPI directory")
endif()
add_library(PulseVideoNative SHARED PulseVideoNative.cpp)
target_compile_features(PulseVideoNative PRIVATE cxx_std_17)
target_compile_definitions(PulseVideoNative PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(PulseVideoNative PRIVATE "${UNITY_PLUGIN_API}")
target_link_libraries(PulseVideoNative PRIVATE d3d11 dxgi mfplat mfreadwrite mfuuid ole32)
if(MSVC)
    set_property(TARGET PulseVideoNative PROPERTY
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(PulseVideoNative PRIVATE /W4 /EHsc /utf-8)
endif()
